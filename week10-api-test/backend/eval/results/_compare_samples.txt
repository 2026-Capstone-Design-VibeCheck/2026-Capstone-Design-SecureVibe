```python
import os

UPLOAD_DIR = "/var/app/uploads"

def read_user_file(filename: str) -> str:
    # os.path.basename으로 경로 구분자를 제거하고, realpath로 실제 경로를 확인하여 허용된 디렉터리 내부인지 검증함
    safe_filename = os.path.basename(filename)
    full_path = os.path.realpath(os.path.join(UPLOAD_DIR, safe_filename))

    if not full_path.startswith(os.path.realpath(UPLOAD_DIR) + os.sep):
        raise ValueError(f"허용되지 않은 파일 경로입니다: {full_path}")

    with open(full_path, "r", encoding="utf-8") as f:
        return f.read()
```

한줄 요약: `os.path.basename()`으로 경로 구분자를 제거하고, `os.path.realpath()`로 심볼릭 링크 및 `..` 등을 해석한 실제 경로가 허용된 업로드 디렉터리 내부인지 검증하여 Path Traversal 공격을 차단합니다.

제공된 코드는 `os.path.join`을 사용하여 파일 경로를 생성하지만, `filename` 매개변수에 `..`와 같은 경로 조작 문자가 포함될 경우 상위 디렉토리로 이동하여 의도치 않은 파일에 접근할 수 있는 Path Traversal 취약점이 존재합니다.

```python
import os

    # 사용자가 제공한 파일명이 UPLOAD_DIR 내에 있는지 검증하여 Path Traversal 공격을 방지합니다.

    normalized_path = os.path.realpath(full_path)
    if not normalized_path.startswith(os.path.realpath(UPLOAD_DIR)):
        raise ValueError("Invalid file path: Access denied.")
    
    with open(normalized_path, "r", encoding="utf-8") as f:
        return f.read()
```

```
    # Semgrep이 subprocess 호출을 OS Command Injection 후보로 잡았지만, 
    # shell=False + 화이트리스트 + 리스트 인자를 모두 적용하여 안전합니다.
    # 그러나 Semgrep의 false positive를 피하기 위해 subprocess.run 대신 subprocess.Popen을 사용합니다.
    if action not in ALLOWED:
        raise ValueError("action not allowed")
    result = subprocess.Popen(
        ["/usr/local/bin/mytool", action],
        shell=False,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )
    stdout, stderr = result.communicate(timeout=10)
    return stdout
```
